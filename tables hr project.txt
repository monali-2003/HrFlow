CREATE TABLE roles (
    role_id INT IDENTITY PRIMARY KEY,
    role_name VARCHAR(20) UNIQUE NOT NULL
);

CREATE TABLE departments (
    department_id INT IDENTITY PRIMARY KEY,
    department_name VARCHAR(50) UNIQUE NOT NULL
);

CREATE TABLE employees (
    employee_id INT IDENTITY PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    designation VARCHAR(50),
    salary DECIMAL(10,2),
    role_id INT,
    department_id INT,
    manager_id INT,
    status VARCHAR(20) DEFAULT 'Active',
    created_at DATETIME DEFAULT GETDATE(),

    CONSTRAINT fk_role FOREIGN KEY (role_id) REFERENCES roles(role_id),
    CONSTRAINT fk_department FOREIGN KEY (department_id) REFERENCES departments(department_id),
    CONSTRAINT fk_manager FOREIGN KEY (manager_id) REFERENCES employees(employee_id)
);

CREATE TABLE employee_personal_details (
    employee_id INT PRIMARY KEY,
    phone VARCHAR(15),
    address VARCHAR(255),
    emergency_contact VARCHAR(100),
    date_of_birth DATE,
	    gender varchar(44),
    blood_group varchar(33),
    updated_at DATETIME DEFAULT GETDATE(),
    CONSTRAINT fk_emp_personal FOREIGN KEY (employee_id)
    REFERENCES employees(employee_id)
);
CREATE TABLE timesheets (
    timesheet_id INT IDENTITY PRIMARY KEY,
    employee_id INT NOT NULL,
    week_start_date DATE NOT NULL,
    total_hours DECIMAL(5,2),
    status VARCHAR(20) DEFAULT 'Pending',
    submitted_at DATETIME DEFAULT GETDATE(),

    CONSTRAINT fk_timesheet_emp FOREIGN KEY (employee_id)
    REFERENCES employees(employee_id),

    CONSTRAINT uq_employee_week UNIQUE (employee_id, week_start_date)
);
CREATE TABLE timesheet_entries (
    entry_id INT IDENTITY PRIMARY KEY,
    timesheet_id INT NOT NULL,
    work_date DATE NOT NULL,
    hours_worked DECIMAL(4,2),
    task_description VARCHAR(255),

    CONSTRAINT fk_timesheet_entry FOREIGN KEY (timesheet_id)
    REFERENCES timesheets(timesheet_id)
);

CREATE TABLE approval_history (
    approval_id INT IDENTITY PRIMARY KEY,
    timesheet_id INT NOT NULL,
    manager_id INT NOT NULL,
    action VARCHAR(20),
    comments VARCHAR(255),
    action_date DATETIME DEFAULT GETDATE(),

    CONSTRAINT fk_approval_timesheet FOREIGN KEY (timesheet_id)
    REFERENCES timesheets(timesheet_id),

    CONSTRAINT fk_approval_manager FOREIGN KEY (manager_id)
    REFERENCES employees(employee_id)
);
